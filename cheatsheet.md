pwd                      # где я сейчас?
cd ~                     # домой
mkdir -p study/week1     # создать папку study и внутри неё week1
cd study/week1           # относительный путь
pwd                      # убедись, где ты
touch a.txt b.txt        # создать два пустых файла
ls -la                   # список с подробностями: видишь . и ..?
echo "привет" > a.txt    # записать текст в файл
cat a.txt                # показать файл
cp a.txt c.txt           # скопировать
mv b.txt renamed.txt     # переименовать
ls -la
cd ..                    # на уровень выше
pwd                      # теперь ты в study
cd ./week1               # то же самое, что cd week1
cd /                     # корень системы, абсолютный путь
ls                       # посмотри на системные папки
cd ~/study/week1         # вернуться одной командой из любого места
rm c.txt                 # удалить файл (без корзины!)
ls -la
