pwd                      # где я сейчас?
cd ~                     # домой
mkdir -p study/week1     # создать папку study и внутри неё week1
cd study/week1           # относительный путь
pwd                      # убедись, где ты
touch a.txt b.txt        # создать два пустых файла
ls -la                   # список с подробностями: видишь . и ..?
echo "привет" > a.txt    # записать текст в файл
cat a.txt                # показать файл
cp a.txt c.txt           # скопировать

mv b.txt renamed.txt     # переименовать
mv — это сокращение от move, «переместить». Переименование — частный случай перемещения.
  Для Linux это одна и та же операция: файл меняет полный путь. Путь состоит из папки и имени:
mv может поменять любую часть пути:
Команда	Что меняется	Итог
  mv cat.jpg dog.jpg	только имя, папка та же	переименование
  mv cat.jpg ../study/	только папка, имя то же	перемещение
  mv cat.jpg ../study/dog.jpg	и папка, и имя	перемещение с переименованием
mv молча перезаписывает. Если в папке назначения уже есть файл с таким же именем, он будет заменён без предупреждения и без корзины. Чтобы Linux спрашивал, добавляй -i:
Если файл там уже есть, появится вопрос overwrite?. Для cp то же самое: cp -i.
